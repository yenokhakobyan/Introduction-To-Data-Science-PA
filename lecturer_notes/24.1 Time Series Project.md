Feature engineering for Bitcoin price forecasting.

import numpy as np
from typing import List, Optional

class FeatureEngineer:

    - Time-based features (hour, day, month)

        self.rolling_windows = [6, 12, 24, 168]  # Rolling windows (hours)

        """
        Create all features from raw OHLCV data.

        df_features = self._create_lag_features(df_features)

        rolling_std = df['close'].rolling(window).std()

"""
Base model class for time series forecasting.

Provides interface that all forecasting models must implement.
"""

from abc import ABC, abstractmethod
import pandas as pd
import numpy as np
from typing import Dict, Tuple, Optional, List
import joblib
import json
from datetime import datetime
import logging

logger = logging.getLogger(__name__)


class BaseForecaster(ABC):
    """
    Abstract base class for all forecasting models.
    
    All models must implement:
    - fit(X_train, y_train)
    - predict(X)
    - get_params()
    - set_params(**params)
    
    Example:
        >>> model = ConcreteForecaster()
        >>> model.fit(X_train, y_train)
        >>> predictions = model.predict(X_test)
    """
    
    def __init__(self, name: str):
        """
        Initialize base forecaster.
        
        Args:
            name: Model name for logging and tracking
        """
        self.name = name
        self.is_fitted = False
        self.feature_names = None
        self.train_metadata = {}
        
    @abstractmethod
    def fit(self, X: pd.DataFrame, y: pd.Series) -> 'BaseForecaster':
        """
        Train the model.
        
        Args:
            X: Features DataFrame
            y: Target Series
        
        Returns:
            self (for method chaining)
        
        TODO for students:
            Each model implementation must:
            1. Validate input data
            2. Train the model
            3. Store feature names
            4. Set is_fitted = True
            5. Store training metadata (date, samples, etc.)
        """
        pass
    
    @abstractmethod
    def predict(self, X: pd.DataFrame) -> np.ndarray:
        """
        Make predictions.
        
        Args:
            X: Features DataFrame
        
        Returns:
            Array of predictions
        
        TODO for students:
            Each model implementation must:
            1. Check that model is fitted
            2. Validate input features match training
            3. Generate predictions
            4. Return as numpy array
        """
        pass
    
    def predict_proba(self, X: pd.DataFrame, 
                     quantiles: List[float] = [0.05, 0.5, 0.95]) -> Dict[float, np.ndarray]:
        """
        Make probabilistic predictions (prediction intervals).
        
        Args:
            X: Features DataFrame
            quantiles: List of quantiles to predict
        
        Returns:
            Dictionary mapping quantile to predictions
        
        Note: This is optional. Models that support probabilistic
        forecasting should override this method.
        """
        raise NotImplementedError(
            f"{self.name} does not support probabilistic predictions"
        )
    
    @abstractmethod
    def get_params(self) -> Dict:
        """
        Get model hyperparameters.
        
        Returns:
            Dictionary of parameter names and values
        """
        pass
    
    @abstractmethod
    def set_params(self, **params) -> 'BaseForecaster':
        """
        Set model hyperparameters.
        
        Args:
            **params: Parameter names and values
        
        Returns:
            self (for method chaining)
        """
        pass
    
    def save(self, filepath: str):
        """
        Save model to disk.
        
        Args:
            filepath: Path to save model file
        
        TODO for students:
            1. Create directory if needed
            2. Save model using joblib
            3. Save metadata as JSON
            4. Log confirmation
        """
        # Save model object
        joblib.dump(self, filepath)
        
        # Save metadata separately for easy access
        metadata_path = filepath.replace('.pkl', '_metadata.json')
        with open(metadata_path, 'w') as f:
            json.dump({
                'name': self.name,
                'is_fitted': self.is_fitted,
                'feature_names': self.feature_names,
                'train_metadata': self.train_metadata,
                'save_date': datetime.now().isoformat()
            }, f, indent=2)
        
        logger.info(f"Model saved to {filepath}")
    
    @classmethod
    def load(cls, filepath: str) -> 'BaseForecaster':
        """
        Load model from disk.
        
        Args:
            filepath: Path to model file
        
        Returns:
            Loaded model instance
        """
        model = joblib.load(filepath)
        logger.info(f"Model loaded from {filepath}")
        return model
    
    def validate_features(self, X: pd.DataFrame):
        """
        Validate that features match training features.
        
        Args:
            X: Features DataFrame
        
        Raises:
            ValueError if features don't match
        
        TODO for students:
            1. Check that all training features are present
            2. Check for unexpected extra features
            3. Raise informative error if mismatch
        """
        if self.feature_names is None:
            logger.warning("No feature names stored, skipping validation")
            return
        
        # Check for missing features
        missing = set(self.feature_names) - set(X.columns)
        if missing:
            raise ValueError(f"Missing features: {missing}")
        
        # Check for extra features
        extra = set(X.columns) - set(self.feature_names)
        if extra:
            logger.warning(f"Extra features will be ignored: {extra}")
    
    def __repr__(self):
        """String representation of model."""
        status = "fitted" if self.is_fitted else "not fitted"
        return f"{self.name} ({status})"

        """Get parameters (none for this simple model)."""
        return {}
    
    def set_params(self, **params) -> 'DummyForecaster':
        """Set parameters (none to set)."""
        return self

            X_val: Optional[pd.DataFrame] = None,

        """
        Fit all base models and learn optimal weights.
        
        Args:
            X: Training features
            y: Training target
            X_val: Validation features (required for optimal weighting)
            y_val: Validation target (required for optimal weighting)
        
        TODO for students:
            1. Fit each base model on training data
            2. If validation data provided, compute validation predictions
            3. Learn weights based on weighting_method:
               - 'equal': weights = 1/n for all models
               - 'inverse_error': weights proportional to 1/error
               - 'optimal': solve optimization problem to minimize error
            4. Store weights and set is_fitted = True
        """
        logger.info(f"Training ensemble of {len(self.models)} models...")
        
        # Fit each base model
        for name, model in self.models:
            logger.info(f"  Fitting {name}...")
            model.fit(X, y)
        
        # Learn weights
        if self.weighting_method == 'equal':
            # Equal weights
            self.weights = np.ones(len(self.models)) / len(self.models)
            
        elif self.weighting_method == 'inverse_error':
            # Weights based on inverse validation error
            if X_val is None or y_val is None:
                raise ValueError("Validation data required for inverse_error weighting")
            
            # TODO: Implement inverse error weighting
            # 1. Get predictions from each model on validation set
            # 2. Compute error (MAE) for each model
            # 3. Set weights proportional to 1/error
            # 4. Normalize weights to sum to 1
            
            errors = []
            for name, model in self.models:
                pred = model.predict(X_val)
                error = np.mean(np.abs(y_val - pred))
                errors.append(error)
            
            # Inverse error (smaller error = higher weight)
            inv_errors = 1.0 / (np.array(errors) + 1e-10)
            self.weights = inv_errors / inv_errors.sum()
            
        elif self.weighting_method == 'optimal':
            # Solve optimization problem for best weights
            if X_val is None or y_val is None:
                raise ValueError("Validation data required for optimal weighting")

                model.predict(X_val) for name, model in self.models

                return np.mean(np.abs(y_val - ensemble_pred))

        """
        Train stacking ensemble using cross-validation.

               - Predict on validation fold (out-of-fold predictions)
            3. Stack all out-of-fold predictions
            4. Train meta-model on out-of-fold predictions
            5. Retrain all base models on full training data
        """

        # Initialize out-of-fold predictions array
        oof_predictions = np.zeros((len(X), len(self.base_models)))

"""
Probabilistic forecasting with uncertainty quantification.

Implements conformal prediction and quantile regression
for distribution-free prediction intervals.
"""

import pandas as pd
import numpy as np
from typing import List, Dict, Tuple, Optional
from sklearn.ensemble import GradientBoostingRegressor

logger = logging.getLogger(__name__)


class ConformalPredictor:
    """
    Conformal prediction for distribution-free prediction intervals.
    
    Provides prediction intervals with guaranteed coverage under
    exchangeability assumption, regardless of model or distribution.
    
    Parameters:
        base_model: Any fitted forecasting model
        confidence_level: Target coverage probability (e.g., 0.90)
    
    TODO for students:
        1. Implement calibrate() to compute nonconformity scores
        2. Implement predict_interval() using calibrated scores
        3. Verify coverage on test set
    
    Example:
        >>> model = RandomForestForecaster()
        >>> model.fit(X_train, y_train)
        >>> conformal = ConformalPredictor(model, confidence_level=0.90)
        >>> conformal.calibrate(X_cal, y_cal)
        >>> lower, upper = conformal.predict_interval(X_test)
    """
    
    def __init__(self, base_model, confidence_level: float = 0.90):
        self.base_model = base_model
        self.confidence_level = confidence_level
        self.calibration_scores = None
        self.quantile_value = None
        
    def calibrate(self, X_cal: pd.DataFrame, y_cal: pd.Series):
        """
        Calibrate prediction intervals using calibration set.
        
        Args:
            X_cal: Calibration features
            y_cal: Calibration targets
        
        TODO for students:
            1. Get predictions on calibration set
            2. Compute nonconformity scores: |y_true - y_pred|
            3. Sort scores
            4. Store for later use
        """
        logger.info("Calibrating conformal predictor...")
        
        # Get predictions
        y_pred_cal = self.base_model.predict(X_cal)
        
        # Compute nonconformity scores (absolute residuals)
        self.calibration_scores = np.abs(y_cal - y_pred_cal)
        
        # Sort scores
        self.calibration_scores = np.sort(self.calibration_scores)
        
        logger.info(f"Calibrated with {len(self.calibration_scores)} samples")
        
        return self
    
    def predict_interval(self, X: pd.DataFrame) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate prediction intervals with guaranteed coverage.
        
        Args:
            X: Features for prediction
        
        Returns:
            Tuple of (lower_bounds, upper_bounds)
        
        TODO for students:
            1. Get point predictions
            2. Compute quantile of calibration scores for coverage level
            3. Create interval: [pred - quantile, pred + quantile]
            4. Return lower and upper bounds
        """
        if self.calibration_scores is None:
            raise ValueError("Must calibrate before predicting intervals")
        
        # Get point predictions
        y_pred = self.base_model.predict(X)
        
        # Compute quantile for desired coverage
        # Formula: ceil((n+1) * (1-alpha)) / n
        n = len(self.calibration_scores)
        alpha = 1 - self.confidence_level
        q_level = np.ceil((n + 1) * (1 - alpha)) / n
        q_level = min(q_level, 1.0)  # Cap at 1.0
        
        # Get quantile of calibration scores
        quantile = np.quantile(self.calibration_scores, q_level)
        
        # Create symmetric interval
        lower = y_pred - quantile
        upper = y_pred + quantile
        
        return lower, upper
    
    def evaluate_coverage(self, X_test: pd.DataFrame, 
                         y_test: pd.Series) -> Dict:
        """
        Evaluate prediction interval coverage on test set.
        
        Args:
            X_test: Test features
            y_test: Test targets
        
        Returns:
            Dictionary with coverage metrics
        
        TODO for students:
            1. Generate prediction intervals
            2. Check what fraction of actuals fall in intervals
            3. Compute interval width statistics
            4. Return evaluation metrics
        """
        lower, upper = self.predict_interval(X_test)
        
        # Check coverage
        in_interval = (y_test >= lower) & (y_test <= upper)
        empirical_coverage = in_interval.mean()
        
        # Interval width statistics
        widths = upper - lower
        mean_width = widths.mean()
        
        # Coverage should be >= target (may exceed due to discreteness)
        coverage_achieved = empirical_coverage >= self.confidence_level
        
        results = {
            'target_coverage': self.confidence_level,
            'empirical_coverage': empirical_coverage,
            'coverage_achieved': coverage_achieved,
            'mean_interval_width': mean_width,
            'median_interval_width': np.median(widths),
            'n_test': len(y_test)
        }
        
        return results

        super().__init__(base_model, confidence_level)
        self.window_size = window_size
        self.decay_rate = decay_rate
        self.online_scores = []
        
    def update(self, X_new: pd.DataFrame, y_new: pd.Series):
        """
        Update calibration with new observations.
        
        TODO for students:
            1. Get prediction for new observation
            2. Compute nonconformity score
            3. Add to sliding window
            4. Remove oldest if window full
        """
        # Get prediction
        y_pred = self.base_model.predict(X_new)
        
        # Compute score
        if isinstance(y_pred, np.ndarray):
            y_pred = y_pred[0]
        if isinstance(y_new, pd.Series):
            y_new = y_new.iloc[0]
        
        score = abs(y_new - y_pred)
        
        # Add to window
        self.online_scores.append(score)
        
        # Maintain window size
        if len(self.online_scores) > self.window_size:
            self.online_scores.pop(0)
        
        return self
    
    def predict_interval_adaptive(self, X: pd.DataFrame) -> Tuple[np.ndarray, np.ndarray]:
        """
        Generate adaptive prediction intervals using weighted scores.
        
        TODO for students:
            1. Create exponential weights for scores
            2. Compute weighted quantile
            3. Return intervals
        """
        if len(self.online_scores) < 10:
            raise ValueError("Need at least 10 online scores")
        
        # Get point prediction
        y_pred = self.base_model.predict(X)
        
        # Create exponential weights (recent scores get higher weight)
        scores = np.array(self.online_scores)
        n = len(scores)
        weights = np.array([self.decay_rate ** (n - 1 - i) for i in range(n)])
        weights = weights / weights.sum()
        
        # Compute weighted quantile
        alpha = 1 - self.confidence_level
        sorted_idx = np.argsort(scores)
        sorted_scores = scores[sorted_idx]
        sorted_weights = weights[sorted_idx]
        cumsum_weights = np.cumsum(sorted_weights)
        
        # Find quantile
        q_idx = np.searchsorted(cumsum_weights, 1 - alpha)
        q_idx = min(q_idx, n - 1)
        quantile = sorted_scores[q_idx]
        
        # Create interval
        lower = y_pred - quantile
        upper = y_pred + quantile
        
        return lower, upper

        lower_q = alpha / 2
        upper_q = 1 - alpha / 2
        
        # Find closest quantiles we have

        upper_quantile = min(self.quantiles, key=lambda x: abs(x - upper_q))
        median_quantile = min(self.quantiles, key=lambda x: abs(x - 0.5))
        
        lower = self.models[lower_quantile].predict(X)
        median = self.models[median_quantile].predict(X)

        return lower, median, upper

        return {
            'target_coverage': confidence_level,
            'empirical_coverage': empirical_coverage,
            'mean_width': widths.mean(),
            'median_width': np.median(widths),
            'n_test': len(y_test)
        }

"""

"""

from datetime import datetime

logger = logging.getLogger(__name__)


class ModelRegistry:
    """
    Manages model versions and metadata.
    
    Provides version control, performance tracking, and
    model lifecycle management.
    
    Example:
        >>> registry = ModelRegistry('models/')
        >>> version = registry.register_model(model, metadata, metrics)
        >>> loaded_model = registry.load_model('RandomForest', version='v1')
    """
    
    def __init__(self, registry_path: str = 'models/'):
        """
        Initialize model registry.
        
        Args:
            registry_path: Directory to store models and metadata
        """
        self.registry_path = Path(registry_path)
        self.registry_path.mkdir(parents=True, exist_ok=True)
        
        # Create subdirectories
        (self.registry_path / 'artifacts').mkdir(exist_ok=True)
        (self.registry_path / 'metadata').mkdir(exist_ok=True)
        
        # Load or create registry index
        self.index_file = self.registry_path / 'registry_index.json'
        self.index = self._load_index()
        
    def _load_index(self) -> Dict:
        """Load registry index or create new one."""

            with open(self.index_file, 'r') as f:
                return json.load(f)
        return {}

        """Save registry index to disk."""
        with open(self.index_file, 'w') as f:
            json.dump(self.index, f, indent=2)

    """

        Args:
            method: 'page_hinkley' or 'adwin'
            delta: Magnitude parameter
            threshold: Detection threshold
        """

        self.delta = delta
        self.threshold = threshold
        self.reset()

        """Reset detector state."""
        self.sum = 0
        self.x_mean = 0
        self.n = 0
        self.min_sum = 0
        self.max_sum = 0
        
    def update(self, value: float) -> bool:
        """
        Update detector with new observation.
        
        Args:
            value: New data point
        
        Returns:
            True if drift detected
        
        TODO for students:
            1. Update running mean
            2. Update cumulative sum
            3. Check if threshold exceeded
            4. Return drift status
        """
        # Update running mean
        self.n += 1
        self.x_mean += (value - self.x_mean) / self.n
        
        # Update cumulative sum
        self.sum += value - self.x_mean - self.delta
        
        # Track extremes
        self.min_sum = min(self.min_sum, self.sum)
        self.max_sum = max(self.max_sum, self.sum)
        
        # Check for drift (positive drift)
        if self.sum - self.min_sum > self.threshold:
            logger.warning(f"Drift detected at observation {self.n}")
            return True
        
        # Check for drift (negative drift)
        if self.max_sum - self.sum > self.threshold:
            logger.warning(f"Drift detected at observation {self.n}")
            return True
        
        return False

        logger.info("Step 2: Validating data quality...")
        validation = self.validator.validate(df)
        if not validation['is_valid']:
            logger.warning("Data quality issues found!")
            for issue in validation['issues']:
                logger.warning(f"  - {issue}")
        
        # Step 3: Feature engineering
        logger.info("Step 3: Engineering features...")
        df = self.engineer.create_features(df)
        df = df.dropna()
        
        # Step 4: Split data
        logger.info("Step 4: Splitting data...")
        train_size = int((1 - self.config['test_size'] - self.config['val_size']) * len(df))
        val_size = int(self.config['val_size'] * len(df))
        
        feature_cols = [col for col in df.columns 
                       if col not in ['close', 'open', 'high', 'low', 'volume']]
        
        X_train = df[feature_cols].iloc[:train_size]
        y_train = df['close'].iloc[:train_size]
        X_val = df[feature_cols].iloc[train_size:train_size+val_size]
        y_val = df['close'].iloc[train_size:train_size+val_size]
        X_test = df[feature_cols].iloc[train_size+val_size:]
        y_test = df['close'].iloc[train_size+val_size:]
        
        logger.info(f"  Train: {len(X_train)}, Val: {len(X_val)}, Test: {len(X_test)}")
        
        # Step 5: Train models
        logger.info("Step 5: Training models...")
        
        # Random Forest
        rf = RandomForestForecaster(n_estimators=100, max_depth=15)
        rf.fit(X_train, y_train)
        
        # XGBoost
        xgb = XGBoostForecaster(n_estimators=100, max_depth=5)
        xgb.fit(X_train, y_train)
        
        # Ensemble
        models = [('rf', rf), ('xgb', xgb)]
        ensemble = WeightedEnsemble(models, weighting_method='optimal')
        ensemble.fit(X_train, y_train, X_val, y_val)
        
        # Step 6: Evaluate
        logger.info("Step 6: Evaluating models...")
        
        pred_rf = rf.predict(X_test)
        pred_xgb = xgb.predict(X_test)
        pred_ensemble = ensemble.predict(X_test)
        
        results_rf = self.evaluator.evaluate(y_test, pred_rf, "RandomForest")
        results_xgb = self.evaluator.evaluate(y_test, pred_xgb, "XGBoost")
        results_ensemble = self.evaluator.evaluate(y_test, pred_ensemble, "Ensemble")
        
        comparison = self.evaluator.compare_models()
        logger.info("\nModel Comparison:")
        logger.info(f"\n{comparison}")
        
        # Step 7: Register best model
        logger.info("Step 7: Registering best model...")
        best_model = ensemble  # or select based on metrics
        
        metadata = {
            'train_samples': len(X_train),
            'n_features': len(feature_cols),
            'data_range': f"{self.config['train_start']} to {self.config['train_end']}"
        }
        
        version = self.registry.register_model(
            model=best_model,
            model_name="BTC_Ensemble",
            metadata=metadata,
            metrics=results_ensemble,
            tags=['production', 'ensemble']
        )
        
        logger.info(f"Model registered as version: {version}")
        logger.info("="*70)
        logger.info("TRAINING COMPLETE!")
        logger.info("="*70)
        
        return version
    
    def run_inference(self, model_name: str = "BTC_Ensemble",
