```python
    def validate_and_clean_data(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Validate and clean the raw data with comprehensive quality checks
        
        This method demonstrates production data quality practices:
        - Schema validation
        - Range checks for numerical values  
        - Handling of missing values
        - Outlier detection and treatment
        """

        try:
            df_clean['dt'] = pd.to_datetime(df_clean['dt'])

        if negative_sales.any():

            df_clean.loc[negative_sales, 'sale_amount'] = 0
        
        # 3. Handle missing values with business logic
        # For sales amount, missing likely means no sale occurred
        df_clean['sale_amount'] = df_clean['sale_amount'].fillna(0)
        df_clean['hours_sale'] = df_clean['hours_sale'].fillna(0)
        
        # For stock count, missing might indicate no stock data available
        df_clean['stock_hour6_22_cnt'] = df_clean['stock_hour6_22_cnt'].fillna(0)

        weather_cols = ['avg_temperature', 'avg_humidity', 'avg_wind_level', 'precpt']
        for col in weather_cols:
            if col in df_clean.columns:
                # Forward fill but limit to avoid propagating old values too far
                df_clean[col] = df_clean.groupby(['store_id', 'city_id'])[col].fillna(method='ffill', limit=24)
                # Back fill remaining
                df_clean[col] = df_clean.groupby(['store_id', 'city_id'])[col].fillna(method='bfill', limit=24)
        
        # 4. Create derived features that will be useful for modeling
        df_clean['hour'] = df_clean['dt'].dt.hour
        df_clean['day_of_week'] = df_clean['dt'].dt.dayofweek  
        df_clean['month'] = df_clean['dt'].dt.month
        df_clean['is_weekend'] = df_clean['day_of_week'].isin([5, 6]).astype(int)
        
        # 5. Remove obvious outliers (extreme values that are likely errors)
        # Define reasonable bounds based on business understanding
        sales_upper_bound = df_clean['sale_amount'].quantile(0.999)
        extreme_sales = df_clean['sale_amount'] > sales_upper_bound
        if extreme_sales.any():
            logger.warning(f"Found {extreme_sales.sum()} records with extreme sales values")
            # In production, you might want to investigate these rather than remove them

        logger.info(f"Data cleaning completed. Rows: {initial_rows} -> {final_rows}")
        
        return df_clean
```

```python
    def run_data_quality_checks(self, df: pd.DataFrame) -> Dict[str, float]:
        """
        Run comprehensive data quality checks and return metrics
        
        These metrics should be monitored in production dashboards
        to catch data quality degradation early
        """
        logger.info("Running data quality checks...")
        
        checks = {}
        
        # Basic completeness checks
        checks['total_records'] = len(df)
        checks['null_percentage_sales'] = (df['sale_amount'].isnull().sum() / len(df)) * 100
        checks['null_percentage_store'] = (df['store_id'].isnull().sum() / len(df)) * 100
        checks['null_percentage_product'] = (df['product_id'].isnull().sum() / len(df)) * 100
        
        # Business logic checks
        checks['negative_sales_percentage'] = ((df['sale_amount'] < 0).sum() / len(df)) * 100
        checks['zero_sales_percentage'] = ((df['sale_amount'] == 0).sum() / len(df)) * 100
        
        # Data distribution checks
        checks['avg_daily_sales'] = df.groupby(df['dt'].dt.date)['sale_amount'].sum().mean()
        checks['unique_stores'] = df['store_id'].nunique()
        checks['unique_products'] = df['product_id'].nunique()
        checks['date_range_days'] = (df['dt'].max() - df['dt'].min()).days
        
        # Consistency checks
        checks['duplicate_records'] = df.duplicated().sum()
        
        # Log critical issues
        if checks['null_percentage_sales'] > 10:
            logger.warning(f"High percentage of null sales: {checks['null_percentage_sales']:.2f}%")
        
        if checks['negative_sales_percentage'] > 1:
            logger.warning(f"High percentage of negative sales: {checks['negative_sales_percentage']:.2f}%")
        
        logger.info("Data quality checks completed")
        return checks
```

```python
import pandas as pd
import numpy as np
from sklearn.preprocessing import StandardScaler, LabelEncoder
from typing import Tuple, List, Dict, Any, Optional
import logging

import os

logger = logging.getLogger(__name__)

class FeatureEngineeringPipeline:
    """
    A comprehensive feature engineering pipeline for time series sales data.
    
    Key principles implemented:
    - Prevent data leakage by only using past information
    - Handle missing values gracefully
    - Create meaningful temporal features
    - Ensure consistency between training and prediction time
    """
    
    def __init__(self, save_path: Optional[str] = None):
        """
        Initialize the feature engineering pipeline
        
        Args:
            save_path: Directory to save fitted transformers
        """
        self.scalers = {}
        self.encoders = {}
        self.save_path = save_path or "models/transformers"

    def create_temporal_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Create time-based features that capture seasonal patterns
        
        Time features are crucial for retail data because:
        - Sales patterns differ by hour of day (morning rush, lunch, evening)
        - Weekly patterns (weekends vs weekdays)
        - Monthly/seasonal patterns (holidays, weather effects)
        """
        logger.info("Creating temporal features...")
        
        df = df.copy()
        
        # Basic time components
        df['hour'] = df['dt'].dt.hour
        df['day_of_week'] = df['dt'].dt.dayofweek
        df['day_of_month'] = df['dt'].dt.day
        df['month'] = df['dt'].dt.month
        df['quarter'] = df['dt'].dt.quarter
        df['year'] = df['dt'].dt.year
        
        # Binary indicators for special time periods
        df['is_weekend'] = df['day_of_week'].isin([5, 6]).astype(int)
        df['is_month_start'] = df['dt'].dt.is_month_start.astype(int)
        df['is_month_end'] = df['dt'].dt.is_month_end.astype(int)
        df['is_quarter_start'] = df['dt'].dt.is_quarter_start.astype(int)
        df['is_quarter_end'] = df['dt'].dt.is_quarter_end.astype(int)
        
        # Cyclical encoding - this preserves the circular nature of time
        # Hour 23 and hour 0 should be "close" to each other
        df['hour_sin'] = np.sin(2 * np.pi * df['hour'] / 24)
        df['hour_cos'] = np.cos(2 * np.pi * df['hour'] / 24)
        df['day_sin'] = np.sin(2 * np.pi * df['day_of_week'] / 7)
        df['day_cos'] = np.cos(2 * np.pi * df['day_of_week'] / 7)
        df['month_sin'] = np.sin(2 * np.pi * df['month'] / 12)
        df['month_cos'] = np.cos(2 * np.pi * df['month'] / 12)
        
        logger.info(f"Created {sum(1 for col in df.columns if col.endswith(('_sin', '_cos', 'hour', 'day_', 'month', 'quarter', 'is_')))} temporal features")
        return df
```

#### 3.2 Implement Lag and Rolling Window Features
These are crucial for time series forecasting as they capture recent trends and patterns.

```python
    def create_lag_features(self, df: pd.DataFrame, 
                          group_cols: List[str], 
                          target_col: str, 
                          lags: List[int]) -> pd.DataFrame:
        """
        Create lag features while preventing data leakage
        
        Lag features use past values to predict future values.
        Critical: We must ensure we only use information available at prediction time!
        
        Args:
            group_cols: Columns to group by (e.g., ['store_id', 'product_id'])
            target_col: Column to create lags for (e.g., 'sale_amount') 
            lags: List of lag periods (e.g., [1, 24, 168] for 1hr, 1day, 1week)
        """
        logger.info(f"Creating lag features for {target_col} with lags {lags}")
        
        df = df.copy()
        # Ensure data is sorted chronologically within each group
        df = df.sort_values(group_cols + ['dt'])
        
        for lag in lags:
            lag_col = f"{target_col}_lag_{lag}"
            # Use groupby to ensure lags are computed within each store-product combination
            df[lag_col] = df.groupby(group_cols)[target_col].shift(lag)
            
        # Count how many lag features we created
        lag_feature_count = len([col for col in df.columns if f"{target_col}_lag_" in col])
        logger.info(f"Created {lag_feature_count} lag features")
        
        return df
    
    def create_rolling_features(self, df: pd.DataFrame, 
                              group_cols: List[str], 
                              target_col: str, 
                              windows: List[int]) -> pd.DataFrame:
        """
        Create rolling window features to capture recent trends
        
        Rolling features smooth out noise and capture trends:
        - Rolling mean shows average recent performance
        - Rolling std shows volatility/stability
        - Rolling min/max show recent extremes
        """
        logger.info(f"Creating rolling features for {target_col} with windows {windows}")
        
        df = df.copy()
        df = df.sort_values(group_cols + ['dt'])
        
        for window in windows:
            # Rolling statistics
            rolling_stats = df.groupby(group_cols)[target_col].rolling(
                window=window, 
                min_periods=1  # Allow computation even with fewer periods
            )
            
            # Reset index to align with original dataframe
            df[f"{target_col}_rolling_mean_{window}"] = (
                rolling_stats.mean().reset_index(level=group_cols, drop=True)
            )
            
            df[f"{target_col}_rolling_std_{window}"] = (
                rolling_stats.std().reset_index(level=group_cols, drop=True)
            )
            
            df[f"{target_col}_rolling_min_{window}"] = (
                rolling_stats.min().reset_index(level=group_cols, drop=True)
            )
            
            df[f"{target_col}_rolling_max_{window}"] = (
                rolling_stats.max().reset_index(level=group_cols, drop=True)
            )
            
            # Fill NaN values that occur at the beginning of each group
            for stat in ['mean', 'std', 'min', 'max']:
                col_name = f"{target_col}_rolling_{stat}_{window}"
                if stat == 'std':
                    # Standard deviation of single value should be 0
                    df[col_name] = df[col_name].fillna(0)
                else:
                    # For other stats, use the current value when no history available
                    df[col_name] = df[col_name].fillna(df[target_col])
        
        rolling_feature_count = len([col for col in df.columns if f"{target_col}_rolling_" in col])
        logger.info(f"Created {rolling_feature_count} rolling features")
        
        return df
```

**💡 Hint**: The `min_periods=1` parameter is crucial. Without it, you'd have NaN values at the beginning of each time series, which could cause problems during training.

**🤔 Reflection Question**: Why do we group by store_id and product_id when creating lag features? What would happen if we didn't group?

```python
    def create_business_features(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Create business-specific features based on retail domain knowledge
        
        These features encode business insights:
        - Stock availability affects sales
        - Promotions drive sales spikes  
        - Weather impacts fresh produce sales
        - Substitution patterns between similar products
        """
        logger.info("Creating business-specific features...")
        
        df = df.copy()
        
        # Stock availability features
        df['is_stocked'] = (df['stock_hour6_22_cnt'] > 0).astype(int)
        df['stock_level_category'] = pd.cut(
            df['stock_hour6_22_cnt'], 
            bins=[-1, 0, 5, 15, float('inf')], 
            labels=['out_of_stock', 'low', 'medium', 'high']
        )
        
        # Promotion intensity features
        df['discount_tier'] = pd.cut(
            df['discount'].fillna(0),
            bins=[-1, 0, 0.1, 0.3, 1.0],
            labels=['no_discount', 'small', 'medium', 'large']
        )
        
        df['has_activity'] = df['activity_flag'].fillna(False).astype(int)
        df['promotion_score'] = df['discount'].fillna(0) + df['has_activity'] * 0.1
        
        # Weather impact features (especially important for fresh produce)
        if 'avg_temperature' in df.columns:
            df['temp_category'] = pd.cut(
                df['avg_temperature'].fillna(df['avg_temperature'].median()),
                bins=[-float('inf'), 10, 25, 35, float('inf')],
                labels=['cold', 'cool', 'warm', 'hot']
            )
            
            # Extreme weather indicator
            temp_median = df['avg_temperature'].median()
            temp_std = df['avg_temperature'].std()
            df['extreme_weather'] = (
                np.abs(df['avg_temperature'] - temp_median) > 2 * temp_std
            ).astype(int)
        
        # Time-based business features
        # Rush hours for food retail
        df['is_rush_hour'] = df['hour'].isin([7, 8, 12, 13, 18, 19]).astype(int)
        
        # Late night/early morning (different shopping patterns)
        df['is_late_night'] = df['hour'].isin([22, 23, 0, 1, 2, 3, 4, 5]).astype(int)
        
        business_feature_count = len([
            col for col in df.columns 
            if any(keyword in col for keyword in ['stock', 'discount', 'promotion', 'temp_', 'weather', 'rush', 'late'])
        ])
        logger.info(f"Created {business_feature_count} business-specific features")
        
        return df
```

**💡 Hint**: Notice how we use domain knowledge to create meaningful bins for continuous variables. These thresholds should come from business understanding, not just statistical analysis.

```python
            experiment_name: Name for the MLflow experiment
        """
        # Set up MLflow
        mlflow.set_experiment(experiment_name)
        self.experiment_name = experiment_name
        self.models = {}
        
        # Create model artifacts directory
        self.model_dir = "models/trained"
        os.makedirs(self.model_dir, exist_ok=True)
        
    def prepare_training_data(self, df: pd.DataFrame, 
                            target_col: str = 'sale_amount') -> Tuple[np.ndarray, np.ndarray, List[str]]:
        """
        Prepare features and target for training with careful feature selection
        
        Feature selection is critical - too many features can lead to overfitting,
        while too few might miss important patterns.
        """
        logger.info("Preparing training data...")
        
        # Define core feature groups
        temporal_features = [
            'hour', 'day_of_week', 'month', 'is_weekend', 'is_rush_hour',
            'hour_sin', 'hour_cos', 'day_sin', 'day_cos', 'month_sin', 'month_cos'
        ]
        
        business_features = [
            'discount', 'holiday_flag', 'activity_flag', 'is_stocked',
            'promotion_score', 'stock_hour6_22_cnt'
        ]
        
        weather_features = [
            'avg_temperature', 'avg_humidity', 'avg_wind_level', 'precpt'
        ]
        
        # Get lag and rolling features that exist in the data
        lag_features = [col for col in df.columns if '_lag_' in col]
        rolling_features = [col for col in df.columns if '_rolling_' in col]
        
        # Combine all feature groups
        feature_cols = temporal_features + business_features + weather_features + lag_features + rolling_features
        
        # Filter to only include columns that actually exist in the dataframe
        available_features = [col for col in feature_cols if col in df.columns]
        
        logger.info(f"Selected {len(available_features)} features for training")
        logger.info(f"Feature categories: Temporal={len([f for f in available_features if f in temporal_features])}, "
                   f"Business={len([f for f in available_features if f in business_features])}, "
                   f"Weather={len([f for f in available_features if f in weather_features])}, "
                   f"Lag={len(lag_features)}, Rolling={len(rolling_features)}")
        
        # Create feature matrix and target vector
        X = df[available_features].copy()
        y = df[target_col].copy()
        
        # Handle missing values (should be minimal after feature engineering)
        X = X.fillna(0)  # Simple strategy, could be made more sophisticated
        
        # Remove rows where target is missing
        mask = ~y.isnull()
        X = X[mask]
        y = y[mask]
        
        logger.info(f"Final dataset shape: {X.shape}, target shape: {y.shape}")
        return X.values, y.values, available_features
    
    def create_time_aware_splits(self, df: pd.DataFrame, n_splits: int = 5) -> List[Tuple[np.ndarray, np.ndarray]]:
        """
        Create time series aware train/validation splits
        
        Critical: For time series, we CANNOT use random splits!
        We must respect temporal order to avoid data leakage.
        """
        logger.info(f"Creating {n_splits} time-aware splits...")
        
        # Sort by datetime to ensure temporal order
        df_sorted = df.sort_values('dt').reset_index(drop=True)
        
        # Use TimeSeriesSplit which respects temporal order
        tscv = TimeSeriesSplit(n_splits=n_splits)
        splits = list(tscv.split(df_sorted))
        
        logger.info(f"Created {len(splits)} time-aware splits")
        return splits
    
    def evaluate_model(self, y_true: np.ndarray, y_pred: np.ndarray) -> Dict[str, float]:
        """
        Calculate comprehensive evaluation metrics for regression
        
        We use multiple metrics because each tells a different story:
        - MAE: Easy to interpret, less sensitive to outliers
        - RMSE: Penalizes large errors more heavily
        - MAPE: Percentage error, good for business understanding
        - R²: Explains variance captured by the model
        """
        # Avoid division by zero in MAPE calculation
        mask = y_true != 0
        mape = mean_absolute_percentage_error(y_true[mask], y_pred[mask]) if mask.any() else np.inf
        
        metrics = {
            'mae': mean_absolute_error(y_true, y_pred),
            'rmse': np.sqrt(mean_squared_error(y_true, y_pred)),
            'mape': mape,
            'r2': r2_score(y_true, y_pred),
            'mean_prediction': np.mean(y_pred),
            'std_prediction': np.std(y_pred),
            'mean_actual': np.mean(y_true),
            'std_actual': np.std(y_true)
        }
        
        return metrics
    
    def train_and_evaluate_models(self, df: pd.DataFrame) -> Dict[str, Any]:
        """
        Train multiple models with time series cross-validation
        
        This method demonstrates production ML best practices:
        - Multiple model comparison
        - Proper time series validation
        - Comprehensive experiment tracking
        - Statistical significance testing
        """
        logger.info("Starting model training and evaluation...")
        
        # Prepare data
        X, y, feature_names = self.prepare_training_data(df)
        splits = self.create_time_aware_splits(df)
        
        # Define model configurations
        models_config = {
            'linear_regression': {
                'model': LinearRegression(),
                'description': 'Simple linear model - good baseline'
            },
            'ridge_regression': {
                'model': Ridge(alpha=1.0, random_state=42),
                'description': 'Regularized linear model - handles multicollinearity'
            },
            'random_forest': {
                'model': RandomForestRegressor(
                    n_estimators=100,
                    max_depth=15,
                    min_samples_split=10,
                    min_samples_leaf=5,
                    random_state=42,
                    n_jobs=-1
                ),
                'description': 'Ensemble method - captures non-linear patterns'
            },
            'gradient_boosting': {
                'model': GradientBoostingRegressor(
                    n_estimators=100,
                    learning_rate=0.1,
                    max_depth=8,
                    min_samples_split=20,
                    min_samples_leaf=10,
                    random_state=42
                ),
                'description': 'Boosting method - often best performer'
            }
        }
        
        results = {}
        best_model_info = None
        best_score = float('inf')
        
        for model_name, config in models_config.items():
            logger.info(f"Training {model_name}...")
            
            # Track this model run in MLflow
            with mlflow.start_run(run_name=f"{model_name}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"):
                
                # Store model configuration
                mlflow.log_param("model_type", model_name)
                mlflow.log_param("model_description", config['description'])
                mlflow.log_param("n_features", len(feature_names))
                mlflow.log_param("n_samples", len(X))
                
                # Perform cross-validation
                cv_scores = {
                    'train_mae': [], 'val_mae': [],
                    'train_rmse': [], 'val_rmse': [],
                    'val_mape': [], 'val_r2': []
                }
                
                fold_models = []
                
                for fold, (train_idx, val_idx) in enumerate(splits):
                    logger.info(f"  Fold {fold + 1}/{len(splits)}")
                    
                    # Split data
                    X_train_fold, X_val_fold = X[train_idx], X[val_idx]
                    y_train_fold, y_val_fold = y[train_idx], y[val_idx]
                    
                    # Train model
                    model = config['model'].__class__(**config['model'].get_params())
                    model.fit(X_train_fold, y_train_fold)
                    fold_models.append(model)
                    
                    # Make predictions
                    train_pred = model.predict(X_train_fold)
                    val_pred = model.predict(X_val_fold)
                    
                    # Evaluate
                    train_metrics = self.evaluate_model(y_train_fold, train_pred)
                    val_metrics = self.evaluate_model(y_val_fold, val_pred)
                    
                    # Store fold results
                    cv_scores['train_mae'].append(train_metrics['mae'])
                    cv_scores['val_mae'].append(val_metrics['mae'])
                    cv_scores['train_rmse'].append(train_metrics['rmse'])
                    cv_scores['val_rmse'].append(val_metrics['rmse'])
                    cv_scores['val_mape'].append(val_metrics['mape'])
                    cv_scores['val_r2'].append(val_metrics['r2'])
                
                # Calculate cross-validation statistics
                cv_stats = {}
                for metric, scores in cv_scores.items():
                    cv_stats[f"{metric}_mean"] = np.mean(scores)
                    cv_stats[f"{metric}_std"] = np.std(scores)
                
                # Log all metrics to MLflow
                mlflow.log_metrics(cv_stats)
                
                # Train final model on all data
                final_model = config['model'].__class__(**config['model'].get_params())
                final_model.fit(X, y)
                
                # Feature importance (if available)
                if hasattr(final_model, 'feature_importances_'):
                    feature_importance = dict(zip(feature_names, final_model.feature_importances_))
                    # Log top 10 most important features
                    top_features = sorted(feature_importance.items(), key=lambda x: x[1], reverse=True)[:10]
                    for i, (feature, importance) in enumerate(top_features):
                        mlflow.log_metric(f"feature_importance_rank_{i+1}", importance)
                        mlflow.log_param(f"top_feature_{i+1}", feature)
                
                # Save model artifact
                model_path = os.path.join(self.model_dir, f"{model_name}_final.joblib")
                joblib.dump(final_model, model_path)
                mlflow.log_artifact(model_path)
                
                # Store results
                results[model_name] = {
                    'model': final_model,
                    'cv_stats': cv_stats,
                    'model_path': model_path,
                    'feature_names': feature_names
                }
                
                # Track best model
                val_mae_mean = cv_stats['val_mae_mean']
                if val_mae_mean < best_score:
                    best_score = val_mae_mean
                    best_model_info = (model_name, final_model, model_path)
                
                logger.info(f"  {model_name} - Val MAE: {val_mae_mean:.4f} ± {cv_stats['val_mae_std']:.4f}")
        
        # Log best model information
        if best_model_info:
            best_name, best_model, best_path = best_model_info
            with mlflow.start_run(run_name="best_model_summary"):
                mlflow.log_param("best_model_name", best_name)
                mlflow.log_metric("best_val_mae", best_score)
                mlflow.log_artifact(best_path)
        
        logger.info(f"Training completed. Best model: {best_name} (MAE: {best_score:.4f})")
        
        return {
            'results': results,
            'best_model': best_model_info,
            'feature_names': feature_names
        }
```

Create `airflow/dags/error_handling_utils.py`:

```python
from airflow.hooks.base_hook import BaseHook
from airflow.models import Variable
import logging
import traceback
from datetime import datetime
from typing import Dict, Any
import json

logger = logging.getLogger(__name__)

class PipelineErrorHandler:
    """
    Centralized error handling and recovery for ML pipeline
    """
    
    def __init__(self):
        self.error_log = []
        
    def log_error(self, task_id: str, error: Exception, context: Dict[str, Any]) -> Dict[str, Any]:
        """
        Log detailed error information for debugging and monitoring
        """
        error_info = {
            'timestamp': datetime.now().isoformat(),
            'task_id': task_id,
            'dag_id': context.get('dag').dag_id,
            'execution_date': context.get('ds'),
            'error_type': type(error).__name__,
            'error_message': str(error),
            'traceback': traceback.format_exc(),
            'task_instance_key': str(context.get('task_instance').key)
        }
        
        self.error_log.append(error_info)
        logger.error(f"Pipeline error in {task_id}: {error_info}")
        
        # Store in Airflow Variable for monitoring dashboard
        try:
            existing_errors = Variable.get('pipeline_errors', default_var='[]', deserialize_json=True)
            existing_errors.append(error_info)
            # Keep only last 50 errors
            recent_errors = existing_errors[-50:]
            Variable.set('pipeline_errors', recent_errors, serialize_json=True)
        except Exception as e:
            logger.warning(f"Failed to store error in Variables: {e}")
            
        return error_info
    
    def attempt_recovery(self, task_id: str, error_type: str, context: Dict[str, Any]) -> bool:
        """
        Attempt automatic recovery based on error type
        """
        recovery_strategies = {
            'ConnectionError': self._recover_connection_error,
            'TimeoutError': self._recover_timeout_error,
            'DataValidationError': self._recover_data_error,
            'ModelTrainingError': self._recover_model_error
        }
        
        if error_type in recovery_strategies:
            try:
                logger.info(f"Attempting recovery for {error_type} in {task_id}")
                return recovery_strategies[error_type](context)
            except Exception as recovery_error:
                logger.error(f"Recovery failed: {recovery_error}")
                return False
        
        return False
    
    def _recover_connection_error(self, context: Dict[str, Any]) -> bool:
        """Recovery strategy for connection errors"""
        # Wait and retry with exponential backoff
        import time
        time.sleep(30)  # Simple wait - in production, use exponential backoff
        return True
    
    def _recover_timeout_error(self, context: Dict[str, Any]) -> bool:
        """Recovery strategy for timeout errors"""
        # Reduce batch size or increase timeout
        context['task_instance'].xcom_push(key='reduce_batch_size', value=True)
        return True
    
    def _recover_data_error(self, context: Dict[str, Any]) -> bool:
        """Recovery strategy for data validation errors"""
        # Skip problematic data and continue with reduced dataset
        context['task_instance'].xcom_push(key='skip_invalid_data', value=True)
        return True
    
    def _recover_model_error(self, context: Dict[str, Any]) -> bool:
        """Recovery strategy for model training errors"""
        # Fall back to simpler model or previous model
        context['task_instance'].xcom_push(key='use_fallback_model', value=True)
        return True

def create_failure_callback(error_handler: PipelineErrorHandler):
    """
    Create a failure callback function for Airflow tasks
    """
    def on_failure_callback(context):
        """
        Called when a task fails - handles error logging and recovery attempts
        """
        task_id = context['task_instance'].task_id
        exception = context.get('exception')
        
        if exception:
            # Log the error
            error_info = error_handler.log_error(task_id, exception, context)
            
            # Attempt recovery
            recovery_attempted = error_handler.attempt_recovery(
                task_id, 
                error_info['error_type'], 
                context
            )
            
            # Send alert to monitoring system
            send_error_alert(error_info, recovery_attempted)
            
        return context
    
    return on_failure_callback

    """
    
    # In production, integrate with:
    # - Slack notifications
    # - PagerDuty for critical errors
    # - Email alerts
    # - Dashboard alerts
```
