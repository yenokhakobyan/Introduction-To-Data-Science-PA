## Apache Airflow - Workflow Orchestration

### What is Airflow?

Apache Airflow is an open-source platform for developing, scheduling, and monitoring workflows. Think of it as the **conductor of your data orchestra** - it ensures all your data processes run in the right order, at the right time, and handles failures gracefully.

### Core Concepts

**DAGs (Directed Acyclic Graphs)**: The heart of Airflow. A DAG represents a workflow with tasks and their dependencies.

**Tasks**: Individual units of work (like running a SQL query, calling an API, or executing a script).

**Operators**: Templates for tasks that define what actually gets executed.

**Scheduler**: The component that monitors DAGs and triggers task execution.

**Executor**: Determines how and where tasks run (locally, on multiple machines, etc.).

### Essential Airflow Functionalities

#### 1. DAG Creation and Management
```python
from airflow import DAG
from airflow.operators.python_operator import PythonOperator
from airflow.operators.bash_operator import BashOperator
from airflow.operators.postgres_operator import PostgresOperator
from datetime import datetime, timedelta

# Default arguments applied to all tasks
default_args = {
    'owner': 'data-team',
    'depends_on_past': False,
    'start_date': datetime(2024, 1, 1),
    'email_on_failure': True,
    'email_on_retry': False,
    'retries': 2,
    'retry_delay': timedelta(minutes=5)
}

# Define the DAG
dag = DAG(
    'my_data_pipeline',
    default_args=default_args,
    description='A comprehensive data pipeline',
    schedule_interval=timedelta(hours=6),  # Run every 6 hours
    catchup=False,  # Don't run historical instances
    tags=['production', 'etl']
)
```

```python
def process_customer_data(**context):
    """Process customer data with business logic"""
    import pandas as pd
    from sqlalchemy import create_engine
    
    # Get execution date from Airflow context
    execution_date = context['execution_date']
    
    # Your business logic here
    engine = create_engine('postgresql://user:pass@localhost/db')
    df = pd.read_sql(f"SELECT * FROM customers WHERE created_date = '{execution_date.date()}'", engine)
    
    # Process data
    processed_count = len(df)
    
    # Return value for downstream tasks
    return processed_count

process_customers = PythonOperator(
```

```python
    dag=dag
)
```

#### 3. Essential DAG Patterns

**ETL Pipeline Template**:
```python
from airflow import DAG
from airflow.operators.dummy_operator import DummyOperator
from airflow.operators.bash_operator import BashOperator
from airflow.operators.python_operator import PythonOperator
from airflow.operators.postgres_operator import PostgresOperator

# Data validation function
def validate_source_data(**context):
    """Validate source data before processing"""
    import psycopg2
    
    conn = psycopg2.connect("postgresql://user:pass@localhost/source_db")
    cur = conn.cursor()
    
    # Check row count
    cur.execute("SELECT COUNT(*) FROM source_table WHERE date = %s", (context['ds'],))
    row_count = cur.fetchone()[0]
    
    if row_count == 0:
        raise ValueError(f"No data found for {context['ds']}")
    
    cur.close()
    conn.close()
    return row_count

# ETL DAG
etl_dag = DAG(
    'daily_etl_pipeline',
    default_args=default_args,
    schedule_interval='0 2 * * *',  # Run at 2 AM daily
    catchup=False
)

# Start task
start = DummyOperator(task_id='start', dag=etl_dag)

# Extract phase
validate_data = PythonOperator(
    task_id='validate_source_data',
    python_callable=validate_source_data,
    dag=etl_dag
)

extract_data = BashOperator(
    task_id='extract_from_source',
    bash_command='python /opt/scripts/extract_data.py --date {{ ds }}',
    dag=etl_dag
)

# Transform phase
transform_data = BashOperator(
    task_id='transform_data',
    bash_command='cd /opt/dbt && dbt run --models staging',
    dag=etl_dag
)

# Load phase
load_data = PostgresOperator(
    task_id='load_to_warehouse',
    postgres_conn_id='warehouse_db',
    sql='/opt/sql/load_transformed_data.sql',
    dag=etl_dag
)

# Data quality checks
quality_check = PythonOperator(
    task_id='final_quality_check',
    python_callable=lambda: print("Quality check passed"),
    dag=etl_dag
)

# End task
end = DummyOperator(task_id='end', dag=etl_dag)

# Define dependencies
start >> validate_data >> extract_data >> transform_data >> load_data >> quality_check >> end
```

#### 4. Advanced Airflow Features

**Dynamic DAG Generation**:
```python
from airflow import DAG
from airflow.operators.postgres_operator import PostgresOperator

# Configuration for multiple similar pipelines
PIPELINE_CONFIG = {
    'sales': {'table': 'sales_data', 'schedule': '0 3 * * *'},
    'marketing': {'table': 'marketing_data', 'schedule': '0 4 * * *'},
    'finance': {'table': 'finance_data', 'schedule': '0 5 * * *'}
}

# Generate DAGs dynamically
for pipeline_name, config in PIPELINE_CONFIG.items():
    dag_id = f'{pipeline_name}_pipeline'
    
    dag = DAG(
        dag_id,
        default_args=default_args,
        schedule_interval=config['schedule'],
        catchup=False,
        tags=[pipeline_name, 'auto-generated']
    )
    
    process_task = PostgresOperator(
        task_id=f'process_{pipeline_name}_data',
        postgres_conn_id='postgres_analytics',
        sql=f"CALL process_{config['table']}('{{{{ ds }}}}');",
        dag=dag
    )
    
    # Make DAG available globally
    globals()[dag_id] = dag
```

**XCom for Task Communication**:
```python
def extract_customer_count(**context):
    """Extract and return customer count"""
    import psycopg2
    
    conn = psycopg2.connect("postgresql://user:pass@localhost/db")
    cur = conn.cursor()
    cur.execute("SELECT COUNT(*) FROM customers WHERE date = %s", (context['ds'],))
    count = cur.fetchone()[0]
    cur.close()
    conn.close()
    
    # Push value to XCom
    return count

def process_if_sufficient(**context):
    """Only process if we have enough customers"""
    # Pull value from XCom
    customer_count = context['task_instance'].xcom_pull(task_ids='extract_count')
    
    if customer_count < 100:
        raise ValueError(f"Insufficient customers: {customer_count}")
    
    print(f"Processing {customer_count} customers")

extract_count = PythonOperator(
    task_id='extract_count',
    python_callable=extract_customer_count,
    dag=dag
)

process_data = PythonOperator(
    task_id='process_data',
    python_callable=process_if_sufficient,
    dag=dag
)

extract_count >> process_data
```

## dbt (Data Build Tool) - Data Transformation

### What is dbt?

dbt (data build tool) is a command-line tool that enables data analysts and engineers to transform data in their warehouse more effectively. Think of dbt as your **data modeling workshop** where raw data gets refined into analytics-ready datasets using SQL and software engineering best practices.

### Core Concepts

**Models**: SQL files that define transformations. Each model is a SELECT statement that creates a table or view.

**Sources**: Raw data tables that dbt reads from but doesn't modify.

**Seeds**: CSV files that dbt can load into your warehouse (useful for lookup tables).

**Snapshots**: Capture point-in-time data for slowly changing dimensions.

**Tests**: Assertions about your data quality (uniqueness, not null, relationships, etc.).

### Essential dbt Functionalities

#### 3. Model Development Patterns

**Staging Models** - Clean and standardize raw data:

**models/staging/stg_customers.sql**:
```sql
{{ config(
    materialized='view',
    tags=['staging', 'customers']
) }}

/*
Staging model for customers - handles data cleaning and standardization.
This model focuses on data quality without complex business logic.
*/

with source_data as (
    select * from {{ source('raw_ecommerce', 'customers') }}
),

cleaned_data as (
    select
        customer_id,
        
        -- Name standardization
        trim(first_name) as first_name,
        trim(last_name) as last_name,
        concat(trim(first_name), ' ', trim(last_name)) as full_name,
        
        -- Email standardization
        lower(trim(email)) as email,
        
        -- Address standardization
        trim(address) as address,
        trim(city) as city,
        upper(trim(state)) as state,
        trim(postal_code) as postal_code,
        upper(trim(country)) as country,
        
        -- Date fields
        registration_date,
        date_of_birth,
        
        -- Calculated fields
        date_part('year', age(date_of_birth)) as customer_age,
        date_part('year', age(registration_date)) as years_as_customer,
        
        -- Customer segmentation
        customer_segment,
        
        -- Metadata
        created_at,
        updated_at
        
    from source_data
    where 
        -- Data quality filters
        email is not null
        and email != ''
        and customer_id is not null
)

select * from cleaned_data
```

**Fact Table Models** - Business events and metrics:

**models/core/fact_orders.sql**:
```sql
{{ config(
    materialized='table',
    indexes=[
      {'columns': ['customer_id'], 'type': 'btree'},
      {'columns': ['order_date'], 'type': 'btree'},
      {'columns': ['order_id'], 'unique': True}
    ]
) }}

/*
Fact table for orders - combines order data with enriched customer and product information.
This table serves as the foundation for order analytics.
*/

with orders as (
    select * from {{ ref('stg_orders') }}
),

customers as (
    select * from {{ ref('stg_customers') }}
),

order_items as (
    select * from {{ ref('stg_order_items') }}
),

products as (
    select * from {{ ref('stg_products') }}
),

-- Calculate order-level metrics
order_metrics as (
    select
        oi.order_id,
        count(*) as item_count,
        sum(oi.quantity) as total_quantity,
        sum(oi.total_price) as items_total,
        avg(p.profit_margin_percent) as avg_product_margin
    from order_items oi
    left join products p on oi.product_id = p.product_id
    group by oi.order_id
),

final as (
    select
        -- Order identifiers
        o.order_id,
        o.customer_id,
        
        -- Customer attributes (denormalized for performance)
        c.customer_segment,
        c.customer_age,
        c.years_as_customer,
        
        -- Order details
        o.order_date,
        o.order_status,
        
        -- Financial metrics
        o.total_amount,
        o.shipping_cost,
        o.tax_amount,
        o.total_amount + o.shipping_cost + o.tax_amount as final_amount,
        
        -- Order composition
        om.item_count,
        om.total_quantity,
        om.avg_product_margin,
        
        -- Calculated metrics
        case 
            when o.total_amount >= 200 then 'High Value'
            when o.total_amount >= 100 then 'Medium Value'
            else 'Low Value'
        end as order_value_tier,
        
        -- Time dimensions
        date_part('year', o.order_date) as order_year,
        date_part('quarter', o.order_date) as order_quarter,
        date_part('month', o.order_date) as order_month,
        date_part('dow', o.order_date) as order_day_of_week,
        
        -- Business day classification
        case 
            when date_part('dow', o.order_date) in (0, 6) then 'Weekend'
            else 'Weekday'
        end as day_type,
        
        -- Payment and shipping
        o.payment_method,
        o.shipping_address,
        
        -- Metadata
        o.created_at,
        o.updated_at
        
    from orders o
    left join customers c on o.customer_id = c.customer_id
    left join order_metrics om on o.order_id = om.order_id
    
    -- Only include completed orders for analytics
    where o.order_status = 'completed'
)

select * from final
```

**Analytical Models** - Business metrics and KPIs:

**models/analytics/monthly_customer_metrics.sql**:
```sql
{{ config(
    materialized='table',
    post_hook="CREATE INDEX IF NOT EXISTS idx_monthly_customer_metrics_month ON {{ this }} (metric_month)"
) }}

/*
Monthly customer metrics for executive reporting.
Provides key customer acquisition, retention, and value metrics by month.
*/

with orders as (
    select * from {{ ref('fact_orders') }}
),

customers as (
    select * from {{ ref('stg_customers') }}
),

-- Monthly order metrics
monthly_orders as (
    select
        date_trunc('month', order_date) as metric_month,
        count(distinct order_id) as total_orders,
        count(distinct customer_id) as active_customers,
        sum(final_amount) as total_revenue,
        avg(final_amount) as avg_order_value,
        
        -- Customer segmentation
        count(distinct case when customer_segment = 'Premium' then customer_id end) as premium_customers,
        count(distinct case when customer_segment = 'Regular' then customer_id end) as regular_customers,
        count(distinct case when customer_segment = 'Budget' then customer_id end) as budget_customers
        
    from orders
    group by date_trunc('month', order_date)
),

-- Monthly customer acquisition
monthly_acquisitions as (
    select
        date_trunc('month', registration_date) as metric_month,
        count(*) as new_customers,
        avg(customer_age) as avg_new_customer_age
    from customers
    group by date_trunc('month', registration_date)
),

-- Customer lifetime value calculations
customer_ltv as (
    select
        customer_id,
        min(order_date) as first_order_date,
        max(order_date) as last_order_date,
        count(*) as total_orders,
        sum(final_amount) as lifetime_value
    from orders
    group by customer_id
),

monthly_ltv as (
    select
        date_trunc('month', first_order_date) as metric_month,
        avg(lifetime_value) as avg_customer_ltv,
        avg(total_orders) as avg_orders_per_customer
    from customer_ltv
    group by date_trunc('month', first_order_date)
),

-- Combine all metrics
final as (
    select
        coalesce(mo.metric_month, ma.metric_month, ml.metric_month) as metric_month,
        
        -- Order metrics
        coalesce(mo.total_orders, 0) as total_orders,
        coalesce(mo.active_customers, 0) as active_customers,
        coalesce(mo.total_revenue, 0) as total_revenue,
        coalesce(mo.avg_order_value, 0) as avg_order_value,
        
        -- Customer segmentation
        coalesce(mo.premium_customers, 0) as premium_customers,
        coalesce(mo.regular_customers, 0) as regular_customers,
        coalesce(mo.budget_customers, 0) as budget_customers,
        
        -- Acquisition metrics
        coalesce(ma.new_customers, 0) as new_customers,
        ma.avg_new_customer_age,
        
        -- LTV metrics
        ml.avg_customer_ltv,
        ml.avg_orders_per_customer,
        
        -- Growth calculations
        lag(mo.total_revenue) over (order by coalesce(mo.metric_month, ma.metric_month, ml.metric_month)) as prev_month_revenue,
        
        case 
            when lag(mo.total_revenue) over (order by coalesce(mo.metric_month, ma.metric_month, ml.metric_month)) > 0 then
                round(
                    ((mo.total_revenue - lag(mo.total_revenue) over (order by coalesce(mo.metric_month, ma.metric_month, ml.metric_month))) 
                     / lag(mo.total_revenue) over (order by coalesce(mo.metric_month, ma.metric_month, ml.metric_month)) * 100)::numeric, 
                    2
                )
            else null
        end as revenue_growth_rate
        
    from monthly_orders mo
    full outer join monthly_acquisitions ma using (metric_month)
    full outer join monthly_ltv ml using (metric_month)
    
    where coalesce(mo.metric_month, ma.metric_month, ml.metric_month) is not null
)

select * from final
order by metric_month
```

## Apache Superset - Data Visualization

### What is Superset?

Apache Superset is a modern data exploration and visualization platform. Think of Superset as your **data storytelling canvas** where you transform numbers into insights through interactive dashboards and compelling visualizations.

### Core Concepts

**Datasets**: Connections to tables, views, or custom SQL queries that serve as data sources for charts.

**Charts**: Individual visualizations (bar charts, line graphs, maps, etc.) built from datasets.

**Dashboards**: Collections of charts arranged to tell a cohesive data story.

**SQL Lab**: Interactive SQL editor for ad-hoc analysis and query development.

```sql
SELECT 
    c.customer_id,

    c.customer_segment,

    SUM(o.final_amount) as total_revenue,
```

```python
# Integration with dbt Cloud for managed dbt
import requests
from airflow.models import Variable

def trigger_dbt_cloud_job(**context):
    """Trigger dbt Cloud job and wait for completion"""
    
    dbt_cloud_token = Variable.get("dbt_cloud_token")
    job_id = Variable.get("dbt_cloud_job_id")
    
    headers = {
        "Authorization": f"Token {dbt_cloud_token}",
        "Content-Type": "application/json"
    }
    
    # Trigger job
    trigger_url = f"https://cloud.getdbt.com/api/v2/accounts/{account_id}/jobs/{job_id}/run/"
    response = requests.post(trigger_url, headers=headers, json={"cause": "Airflow trigger"})
    
    run_id = response.json()["data"]["id"]
    
    # Poll for completion
    import time
    while True:
        status_url = f"https://cloud.getdbt.com/api/v2/accounts/{account_id}/runs/{run_id}/"
        status_response = requests.get(status_url, headers=headers)
        status = status_response.json()["data"]["status"]
        
        if status in ["success", "error", "cancelled"]:
            break
        time.sleep(30)
    
    if status != "success":
        raise Exception(f"dbt job failed with status: {status}")
    
    return run_id
```

```python
        Revenue Anomaly Detected!
        
        Latest Month Revenue: ${recent_revenue:,.2f}
        Historical Average: ${historical_mean:,.2f}
        Z-Score: {z_score:.2f}
        
        This requires investigation.
        """
        send_alert(alert_message)
    
    return z_score
```
